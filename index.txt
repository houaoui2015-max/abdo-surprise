  <script>
    let loveChoice = "";

    function chooseLove(choice){
      loveChoice = choice;
      document.getElementById("step1").style.display = "none";
      document.getElementById("step2").style.display = "block";
      document.getElementById("loveAnswerText").innerHTML =
        "You chose: <b>" + loveChoice + "</b> 😍";
    }

    // Moving NO button
    const noBtn = document.getElementById("noBtn");
    noBtn.addEventListener("mouseenter", moveNo);
    noBtn.addEventListener("click", moveNo);
    noBtn.addEventListener("touchstart", moveNo);

    function moveNo(){
      const maxX = 220;
      const maxY = 120;
      const x = (Math.random() * maxX) - (maxX/2);
      const y = (Math.random() * maxY) - (maxY/2);
      noBtn.style.transform = `translate(${x}px, ${y}px)`;
    }

    // Cute pop sound (no file needed)
    function playPop(){
      try{
        const AudioContext = window.AudioContext || window.webkitAudioContext;
        const ctx = new AudioContext();
        const o = ctx.createOscillator();
        const g = ctx.createGain();

        o.type = "triangle";
        o.frequency.value = 520;
        g.gain.value = 0.08;

        o.connect(g);
        g.connect(ctx.destination);

        o.start();
        o.frequency.exponentialRampToValueAtTime(220, ctx.currentTime + 0.12);
        g.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + 0.18);

        setTimeout(() => {
          o.stop();
          ctx.close();
        }, 220);
      }catch(e){
        // ignore if blocked
      }
    }

    // Hearts/confetti burst
    function makeBurst(){
      const burst = document.getElementById("burst");
      const emojis = ["💘","💖","💗","💕","✨","🎉","🥳","🎊"];
      const count = 22;

      for(let i=0;i<count;i++){
        const el = document.createElement("div");
        el.className = "piece";
        el.textContent = emojis[Math.floor(Math.random()*emojis.length)];

        const x = 50 + (Math.random()*10 - 5); // center-ish
        const y = 55 + (Math.random()*10 - 5);

        el.style.left = x + "vw";
        el.style.top = y + "vh";

        const dx = (Math.random()*600 - 300) + "px";
        const dy = (Math.random()*-520 - 120) + "px";
        el.style.setProperty("--dx", dx);
        el.style.setProperty("--dy", dy);

        el.style.fontSize = (18 + Math.random()*18) + "px";

        burst.appendChild(el);

        setTimeout(() => el.remove(), 1400);
      }
    }

    // YES click
    document.getElementById("yesBtn").addEventListener("click", () => {
      playPop();
      makeBurst();

      document.getElementById("successBox").style.display = "block";
      document.getElementById("celebrate").style.display = "block";
      document.getElementById("photoWrap").style.display = "block";
      noBtn.style.display = "none";
    });
