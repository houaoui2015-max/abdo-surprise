  <script>
    // ======= NAV =======
    const pages = ["page-q1","page-q2","page-yes","page-no"];
    function show(id){
      for(const p of pages) document.getElementById(p).classList.add("hidden");
      document.getElementById(id).classList.remove("hidden");
    }
    function goHome(){
      show("page-q1");
      document.getElementById("q2note").textContent = "";
    }

    // ======= Q1 =======
    let loveChoice = "";
    function pickLove(choice){
      loveChoice = choice;
      // cute note based on answer
      const note = (choice === "bzzf")
        ? "😍 Aww! Okay… now the real question 😄"
        : "😊 Hta shwiya kaykfi باش nفرح… and one more question 😄";
      document.getElementById("q2note").textContent = note;
      show("page-q2");
      popHearts(choice === "bzzf" ? 14 : 10);
    }

    // ======= Q2 =======
    function valYes(){
      show("page-yes");
      burstConfetti();
      popHearts(18);
    }
    function valNo(){
      show("page-no");
      popHearts(8);
    }

    // ======= COPY =======
    async function copyMsg(){
      const text =
        "Kbida dyali 💖 Thank you for saying YES 😍💘 Can’t wait for our Valentine’s ✨";
      try{
        await navigator.clipboard.writeText(text);
        document.getElementById("copiedNote").style.display = "block";
        setTimeout(()=> document.getElementById("copiedNote").style.display="none", 1800);
      }catch(e){
        alert("Copy not supported 😅 Just type: " + text);
      }
    }

    // ======= HEARTS =======
    const heartsLayer = document.getElementById("hearts");
    function spawnHeart(){
      const h = document.createElement("div");
      h.className = "heart";
      h.textContent = Math.random() > 0.5 ? "💖" : "💘";
      const left = Math.random() * 100;
      const size = 14 + Math.random() * 18;
      const duration = 4 + Math.random() * 4;
      const drift = (-40 + Math.random()*80).toFixed(0) + "px";

      h.style.left = left + "vw";
      h.style.fontSize = size + "px";
      h.style.animationDuration = duration + "s";
      h.style.setProperty("--drift", drift);

      heartsLayer.appendChild(h);
      setTimeout(()=> h.remove(), duration*1000);
    }
    setInterval(spawnHeart, 320);
    function popHearts(n=10){
      for(let i=0;i<n;i++) setTimeout(spawnHeart, i*40);
    }

    // ======= CONFETTI =======
    const canvas = document.getElementById("confetti");
    const ctx = canvas.getContext("2d");
    let W,H;
    function resize(){
      W = canvas.width = window.innerWidth;
      H = canvas.height = window.innerHeight;
    }
    window.addEventListener("resize", resize);
    resize();

    let confetti = [];
    function burstConfetti(){
      confetti = [];
      const count = 180;
      for(let i=0;i<count;i++){
        confetti.push({
          x: W/2, y: H/2,
          vx: (Math.random()-0.5)*10,
          vy: (Math.random()-0.9)*12,
          g: 0.22 + Math.random()*0.08,
          s: 3 + Math.random()*4,
          a: 1
        });
      }
    }
    function drawConfetti(){
      ctx.clearRect(0,0,W,H);
      for(const p of confetti){
        p.vy += p.g; p.x += p.vx; p.y += p.vy; p.a *= 0.985;
        ctx.globalAlpha = p.a;
        ctx.fillRect(p.x, p.y, p.s, p.s);
      }
      ctx.globalAlpha = 1;
      confetti = confetti.filter(p => p.a > 0.05 && p.y < H+40);
      requestAnimationFrame(drawConfetti);
    }
    drawConfetti();
